async function doSearch() { const q = document.getElementById('q').value.trim(); if (!q) return; Swal.fire({ title: 'Testing connection...', allowOutsideClick: false, didOpen: () => Swal.showLoading() }); // Pulls raw data without SQL filtering to verify connectivity const testUrl = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/gviz/tq?gid=${GID}&tqx=out:json`; try { const response = await fetch(testUrl); const text = await response.text(); // Extract JSON from Google's wrapper function const jsonMatch = text.match(/google\.visualization\.Query\.setResponse\(([\s\S]*)\);/); if (!jsonMatch) { Swal.fire({ icon: 'error', title: 'Connection Blocked', text: 'Google Sheets returned an unreadable response. Check tab sharing.' }); return; } const gvizData = JSON.parse(jsonMatch[1]); if (gvizData.status === "error") { Swal.fire({ icon: 'error', title: 'Google Sheet Error', text: gvizData.errors[0].detailed_message }); return; } const allRows = gvizData.table.rows; console.log("Raw sheet rows fetched:", allRows); if (!allRows || allRows.length === 0) { Swal.fire({ icon: 'warning', title: 'Sheet is Empty', text: 'Connected to the sheet, but no rows were found on this tab (gid).' }); return; } // Search locally in JavaScript across Column N (Index 13) const search = q.toLowerCase(); const matchedRows = []; allRows.forEach(r => { if (!r.c) return; // Extract plain text values across columns const rowVals = r.c.map(cell => (cell ? (cell.f || cell.v || "") : "")); // Column N is index 13 const teacher = String(rowVals[13] || "").toLowerCase(); if (teacher.includes(search)) { // Mapping indices: C=2, D=3, E=4, F=5, S=18, T=19, R=17, N=13, M=12, O=14, U=20, G=6, H=7, I=8, J=9, K=10, L=11 matchedRows.push([ formatCellDate(rowVals[2]), rowVals[3], rowVals[4], rowVals[5], rowVals[18], rowVals[19], rowVals[17], rowVals[13], rowVals[12], rowVals[14], rowVals[20], rowVals[6], rowVals[7], rowVals[8], rowVals[9], rowVals[10], rowVals[11] ]); } }); if (matchedRows.length === 0) { Swal.fire({ icon: 'info', title: 'Connected successfully!', text: `Connected to tab ${GID} (${allRows.length} total rows found), but no rows match "${q}" in Column N.` }); document.getElementById('results').innerHTML = ""; return; } Swal.close(); render(matchedRows); } catch (err) { console.error("Fetch Error:", err); Swal.fire({ icon: 'error', title: 'Access Denied / Network Error', text: 'Make sure your Google Sheet access is set to "Anyone with the link can view".' }); } }
